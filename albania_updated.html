async function loadAlbaniaData() { const csvUrl = "https://docs.google.com/spreadsheets/d/e/2PACX-1vT3NsJbFLtYWZSBn0WnfBYfO9CBqZfnMW4KIF9oT8DTzmvGX5SyQ03aaBeAMb0L_A0Cqdw81k679x5X/pub?output=csv"; const response = await fetch(csvUrl); const csvText = await response.text(); Papa.parse(csvText, { header: true, skipEmptyLines: true, transformHeader: header => header.trim(), transform: value => value.trim(), complete: function(results) { const data = results.data; const container = document.getElementById("video-container"); data.forEach(row => { if (row.Country !== "Albania") return; const year = row.Year; const artist = row.Artist; const song = row.Song; const videoUrl = row.Video_URL; const videoItem = document.createElement("div"); videoItem.className = "video-item"; videoItem.setAttribute("data-artist", artist); videoItem.setAttribute("data-country", "Albania"); videoItem.setAttribute("data-song", song); videoItem.setAttribute("data-year", year); videoItem.setAttribute("onclick", `GDriveOpenModalWithVideo('${videoUrl}')`); const previewContainer = document.createElement("div"); previewContainer.className = "preview-container"; const img = document.createElement("img"); img.src = `images/${year}-AL.jpeg`; img.alt = `Albania ${year}`; previewContainer.appendChild(img); const titleWithIcon = document.createElement("div"); titleWithIcon.className = "video-title-with-icon"; titleWithIcon.setAttribute("onclick", "event.stopPropagation()"); const a = document.createElement("a"); a.href = "albania.html"; const flagImg = document.createElement("img"); flagImg.src = "flags/al.png"; flagImg.className = "icon-placeholder"; flagImg.alt = "Flag"; a.appendChild(flagImg); titleWithIcon.appendChild(a); const titleText = document.createElement("div"); titleText.className = "video-title-text"; titleText.innerHTML = `${artist}
${song}
${year}`; titleWithIcon.appendChild(titleText); videoItem.appendChild(previewContainer); videoItem.appendChild(titleWithIcon); container.appendChild(videoItem); }); } }); }
